{{- /* Images are resolved from assets/images/ by filename. Keeping them there (rather than beside the post) means the multi-megabyte originals are never published — only the generated webp files ship. */ -}} {{- $name := path.Base .Destination -}} {{- $img := resources.Get (printf "images/%s" $name) -}} {{- if $img -}} {{- $w := $img.Width -}} {{- if gt $w 1400 -}} {{- $lg := $img.Resize "1400x webp q82" -}} {{- $sm := $img.Resize "760x webp q78" -}} {{- else -}} {{- /* Never upscale. Most screenshots are narrower than 1400px, and resizing those up would blur them and inflate the file for no gain. */ -}} {{- $one := $img.Resize (printf "%dx webp q82" $w) -}} {{- end -}} {{- else -}} {{- warnf "render-image: no file at assets/images/%s (referenced from %s) — the below will 404" $name .Page.File.Path -}} {{- end -}}
