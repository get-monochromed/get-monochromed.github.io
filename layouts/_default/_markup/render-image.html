{{- /* Images are resolved from assets/images/ by filename. Keeping them there (rather than beside the post) means the multi-megabyte originals are never published — only the generated webp files ship. */ -}} {{- $name := path.Base .Destination -}} {{- $img := resources.Get (printf "images/%s" $name) -}} {{- if $img -}} {{- $lg := $img.Resize "1400x webp q82" -}} {{- $sm := $img.Resize "760x webp q78" -}} {{- else -}} {{- warnf "render-image: no file at assets/images/%s (referenced from %s) — the below will 404" $name .Page.File.Path -}} {{- end -}}
